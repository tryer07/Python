修改表名：

ALTER TABLE 旧表名 RENAME TO 新表名;
-- 举例：
ALTER TABLE employees RENAME TO emp;

修改列名（重命名列）：

-- 方法一：CHANGE（老写法，必须带上原数据类型，不推荐）
ALTER TABLE 表名 CHANGE COLUMN 旧列名 新列名 数据类型;

-- 方法二：RENAME COLUMN（MySQL 8.0 推荐写法，不需要带类型，更安全）
ALTER TABLE 表名 RENAME COLUMN 旧列名 TO 新列名;
-- 举例：
ALTER TABLE emp RENAME COLUMN employee_name TO emp_name;

修改列类型：

ALTER TABLE 表名 MODIFY COLUMN 列名 新数据类型;
-- 举例（把工资的精度改大一点）：
ALTER TABLE emp MODIFY COLUMN salary DECIMAL(10,2);

添加新列：

ALTER TABLE 表名 ADD COLUMN 新列名 数据类型;
-- 举例（默认加在最后面）：
ALTER TABLE emp ADD COLUMN age INT;
-- 进阶：指定加在某个列后面（AFTER）或者加在最前面（FIRST）
ALTER TABLE emp ADD COLUMN gender CHAR(1) AFTER emp_name;

删除指定列：

ALTER TABLE 表名 DROP COLUMN 列名;
-- 举例：
ALTER TABLE emp DROP COLUMN age;
